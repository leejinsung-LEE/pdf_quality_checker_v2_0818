# PDF Quality Checker v2.0 - 확장형 프로젝트 설계 명세서

## 프로젝트 개요

### 목적
PDF 파일의 인쇄 품질을 자동으로 검사하고 문제점을 수정하는 데스크톱 애플리케이션으로, 향후 판짜기(Imposition) 기능 확장을 고려한 설계

### 핵심 요구사항
- PDF 품질 분석 (폰트, 색상, 이미지, 인쇄 설정)
- 자동 문제 수정 (RGB→CMYK, 폰트 임베딩)
- 다중 폴더 실시간 감시
- 프로파일 기반 검사 설정
- 상세 보고서 생성
- **[향후]** 인쇄 판짜기 통합 지원

### 기술 스택
- Python 3.10+
- GUI: CustomTkinter (모던 UI)
- PDF 처리: pikepdf, PyMuPDF (fitz)
- 외부 도구: Ghostscript, pdffonts (poppler)
- **[향후]** pdftk, pdfnup (판짜기용)

```python
from dataclasses import dataclass, field
from pathlib import Path
from typing import List, Dict, Optional
from datetime import datetime

@dataclass
class PDFDocument:
    """PDF 문서의 핵심 정보를 담는 불변 객체"""
    path: Path
    file_hash: str
    file_size: int
    created_at: datetime
    
    # 분석 후 추가되는 정보
    page_count: int = 0
    pdf_version: str = ""
    is_encrypted: bool = False
    metadata: Dict[str, str] = field(default_factory=dict)
    
    # 판짜기 관련 준비 정보
    is_uniform_size: bool = True  # 모든 페이지가 같은 크기인지
    dominant_page_size: Optional['PageSize'] = None  # 가장 많이 사용된 페이지 크기
    
    def __post_init__(self):
        """생성 시 유효성 검증"""
        if not self.path.exists():
            raise ValueError(f"PDF 파일이 존재하지 않습니다: {self.path}")
        if not self.path.suffix.lower() == '.pdf':
            raise ValueError(f"PDF 파일이 아닙니다: {self.path}")

@dataclass
class PageInfo:
    """개별 페이지 정보"""
    page_number: int
    width_mm: float
    height_mm: float
    rotation: int = 0
    has_transparency: bool = False
    has_overprint: bool = False
    bleed_info: Optional['BleedInfo'] = None
    
    # 판짜기 관련 정보
    content_bounds: Optional['Rectangle'] = None  # 실제 컨텐츠가 있는 영역
    can_be_rotated: bool = True  # 판짜기 시 회전 가능 여부
    
    @property
    def paper_size(self) -> str:
        """용지 규격 판별 (A4, A3 등)"""
        # 구현 코드
        pass
    
    @property
    def orientation(self) -> str:
        """페이지 방향 (portrait/landscape)"""
        return "portrait" if self.height_mm > self.width_mm else "landscape"

@dataclass
class Rectangle:
    """사각형 영역 정의 (판짜기용)"""
    x: float
    y: float
    width: float
    height: float
    
    def contains(self, other: 'Rectangle') -> bool:
        """다른 사각형을 포함하는지 확인"""
        return (self.x <= other.x and 
                self.y <= other.y and
                self.x + self.width >= other.x + other.width and
                self.y + self.height >= other.y + other.height)

@dataclass
class PageSize:
    """페이지 크기 정보"""
    width_mm: float
    height_mm: float
    name: str  # "A4", "Letter", "Custom" 등
    
    def __eq__(self, other):
        """크기 비교 (1mm 오차 허용)"""
        return (abs(self.width_mm - other.width_mm) < 1 and 
                abs(self.height_mm - other.height_mm) < 1)

@dataclass
class BleedInfo:
    """재단선 정보"""
    top: float = 0.0
    bottom: float = 0.0
    left: float = 0.0
    right: float = 0.0
    
    @property
    def has_proper_bleed(self) -> bool:
        """표준 재단선(3mm) 확인"""
        return all(value >= 3.0 for value in [self.top, self.bottom, self.left, self.right])
    
    @property
    def is_uniform(self) -> bool:
        """모든 방향의 재단선이 동일한지"""
        return len(set([self.top, self.bottom, self.left, self.right])) == 1
```

#### analysis_result.py (판짜기 대비 확장)
```python
from dataclasses import dataclass, field
from typing import List, Dict, Optional
from .pdf_document import PDFDocument, PageInfo
from .quality_issue import QualityIssue
from .imposition import ImpositionReadiness

@dataclass
class AnalysisResult:
    """PDF 분석 결과를 담는 종합 객체"""
    document: PDFDocument
    pages: List[PageInfo]
    fonts: Dict[str, 'FontInfo']
    colors: 'ColorInfo'
    images: List['ImageInfo']
    issues: List[QualityIssue] = field(default_factory=list)
    
    # 분석 메타데이터
    analysis_duration: float = 0.0
    analyzer_version: str = "2.0"
    profile_used: str = "default"
    
    # 판짜기 준비 상태
    imposition_readiness: Optional[ImpositionReadiness] = None
    
    @property
    def has_errors(self) -> bool:
        """오류 존재 여부"""
        return any(issue.severity == 'error' for issue in self.issues)
    
    @property
    def quality_score(self) -> float:
        """품질 점수 계산 (0-100)"""
        if not self.issues:
            return 100.0
        
        deductions = {
            'error': 10,
            'warning': 3,
            'info': 1
        }
        
        total_deduction = sum(
            deductions.get(issue.severity, 0) 
            for issue in self.issues
        )
        
        return max(0, 100 - total_deduction)
    
    @property
    def is_imposition_ready(self) -> bool:
        """판짜기 가능 상태인지 확인"""
        if not self.imposition_readiness:
            return False
        return self.imposition_readiness.is_ready
    
    def get_issues_by_category(self, category: str) -> List[QualityIssue]:
        """카테고리별 이슈 필터링"""
        return [issue for issue in self.issues if issue.category == category]
```

#### imposition.py (향후 구현을 위한 모델)

```python
from dataclasses import dataclass, field
from typing import List, Dict, Optional, Tuple
from enum import Enum

class BindingMethod(Enum):
    """제본 방식"""
    SADDLE_STITCH = "saddle"  # 중철
    PERFECT_BIND = "perfect"   # 무선철
    SPIRAL = "spiral"          # 스프링
    NONE = "none"              # 제본 없음

@dataclass
class ImpositionTemplate:
    """판짜기 템플릿 정보"""
    name: str  # "2-up", "4-up", "8-up saddle" 등
    sheet_width: float  # 용지 너비 (mm)
    sheet_height: float  # 용지 높이 (mm)
    rows: int  # 행 수
    columns: int  # 열 수
    gutter: float = 0.0  # 페이지 간격
    binding_method: BindingMethod = BindingMethod.NONE
    
    # 재단선 및 마크 설정
    crop_marks: bool = True
    registration_marks: bool = True
    color_bars: bool = True
    page_info: bool = True
    
    # 레이아웃 옵션
    auto_rotate: bool = True  # 용지 활용도를 위한 자동 회전
    duplex: bool = True  # 양면 인쇄
    
    @property
    def pages_per_sheet(self) -> int:
        """한 장에 들어가는 페이지 수"""
        return self.rows * self.columns
    
    @property
    def pages_per_signature(self) -> int:
        """한 대(signature)의 페이지 수"""
        if self.binding_method == BindingMethod.SADDLE_STITCH:
            return self.pages_per_sheet * 2 if self.duplex else self.pages_per_sheet
        return self.pages_per_sheet

@dataclass
class ImpositionReadiness:
    """판짜기 준비 상태 정보"""
    is_ready: bool = False
    issues: List[str] = field(default_factory=list)
    warnings: List[str] = field(default_factory=list)
    
    # 페이지 정보
    total_pages: int = 0
    uniform_page_size: bool = True
    page_sizes: List[Tuple[float, float]] = field(default_factory=list)
    
    # 재단선 정보
    all_pages_have_bleed: bool = False
    minimum_bleed: float = 0.0
    
    # 추천 템플릿
    suggested_templates: List[ImpositionTemplate] = field(default_factory=list)
    
    def add_issue(self, message: str):
        """문제 추가"""
        self.issues.append(message)
        self.is_ready = False
    
    def add_warning(self, message: str):
        """경고 추가"""
        self.warnings.append(message)

@dataclass
class ImpositionResult:
    """판짜기 결과"""
    template_used: ImpositionTemplate
    total_sheets: int
    sheet_layouts: List['SheetLayout']  # 각 시트의 레이아웃
    waste_percentage: float  # 용지 낭비율
    blank_pages_added: int  # 추가된 빈 페이지 수
    
    @property
    def total_impressions(self) -> int:
        """총 인쇄면 수"""
        return self.total_sheets * (2 if self.template_used.duplex else 1)

@dataclass
class SheetLayout:
    """개별 시트의 레이아웃"""
    sheet_number: int
    side: str  # "front" or "back"
    page_positions: List['PagePosition']  # 페이지 배치 정보
    
@dataclass
class PagePosition:
    """시트 내 페이지 위치"""
    page_number: int  # 원본 페이지 번호
    x: float  # 시트 내 X 좌표
    y: float  # 시트 내 Y 좌표
    rotation: int  # 회전 각도 (0, 90, 180, 270)
    scale: float = 1.0  # 크기 조정 비율
```

### 2. 분석 엔진 확장 (src/core/analyzers/)

#### imposition_readiness_analyzer.py
```python
from typing import Dict, Any, List
from collections import Counter
from .base_analyzer import BaseAnalyzer
from ..models import PDFDocument, ImpositionReadiness, ImpositionTemplate, BindingMethod

class ImpositionReadinessAnalyzer(BaseAnalyzer):
    """판짜기 준비 상태 분석기"""
    
    def __init__(self):
        super().__init__("ImpositionReadinessAnalyzer")
        self.standard_templates = self._load_standard_templates()
    
    def analyze(self, document: PDFDocument, context: Dict[str, Any]) -> Dict[str, Any]:
        """판짜기 가능성 분석"""
        readiness = ImpositionReadiness(total_pages=document.page_count)
        
        # 분석 컨텍스트에서 페이지 정보 가져오기
        pages_info = context.get('pages', [])
        
        # 1. 페이지 크기 일관성 확인
        self._check_page_size_consistency(pages_info, readiness)
        
        # 2. 재단선 확인
        self._check_bleed_status(pages_info, readiness)
        
        # 3. 페이지 수 확인
        self._check_page_count(document.page_count, readiness)
        
        # 4. 추천 템플릿 생성
        if readiness.uniform_page_size and readiness.page_sizes:
            readiness.suggested_templates = self._suggest_templates(
                readiness.page_sizes[0], 
                document.page_count
            )
        
        # 최종 준비 상태 결정
        readiness.is_ready = len(readiness.issues) == 0
        
        return {
            'imposition_readiness': readiness,
            'can_impose': readiness.is_ready,
            'suggested_templates': [t.name for t in readiness.suggested_templates[:3]]
        }
    
    def _check_page_size_consistency(self, pages_info: List[Dict], readiness: ImpositionReadiness):
        """페이지 크기 일관성 확인"""
        if not pages_info:
            readiness.add_issue("페이지 정보를 찾을 수 없습니다")
            return
        
        # 페이지 크기 수집
        size_counter = Counter()
        for page in pages_info:
            size = (round(page['width_mm'], 1), round(page['height_mm'], 1))
            size_counter[size] += 1
            readiness.page_sizes.append(size)
        
        # 가장 많이 사용된 크기
        most_common_size, count = size_counter.most_common(1)[0]
        
        if len(size_counter) == 1:
            readiness.uniform_page_size = True
        else:
            readiness.uniform_page_size = False
            readiness.add_issue(f"페이지 크기가 일정하지 않습니다. {len(size_counter)}개의 다른 크기 발견")
            
            # 90% 이상이 같은 크기면 경고만
            if count / len(pages_info) >= 0.9:
                readiness.add_warning(f"대부분({count}/{len(pages_info)})의 페이지가 {most_common_size}mm입니다")
    
    def _check_bleed_status(self, pages_info: List[Dict], readiness: ImpositionReadiness):
        """재단선 상태 확인"""
        pages_with_bleed = 0
        min_bleed = float('inf')
        
        for page in pages_info:
            bleed_info = page.get('bleed_info')
            if bleed_info and bleed_info.get('has_proper_bleed'):
                pages_with_bleed += 1
                min_bleed = min(min_bleed, 
                              bleed_info['top'], 
                              bleed_info['bottom'],
                              bleed_info['left'], 
                              bleed_info['right'])
        
        readiness.all_pages_have_bleed = pages_with_bleed == len(pages_info)
        readiness.minimum_bleed = min_bleed if min_bleed != float('inf') else 0
        
        if not readiness.all_pages_have_bleed:
            readiness.add_warning(f"{len(pages_info) - pages_with_bleed}개 페이지에 재단선이 없습니다")
    
    def _check_page_count(self, page_count: int, readiness: ImpositionReadiness):
        """페이지 수 확인"""
        # 중철 제본을 위한 4의 배수 확인
        if page_count % 4 != 0:
            readiness.add_warning(f"중철 제본을 위해서는 4의 배수 페이지가 필요합니다. 현재: {page_count}페이지")
    
    def _suggest_templates(self, page_size: Tuple[float, float], page_count: int) -> List[ImpositionTemplate]:
        """적합한 템플릿 추천"""
        suggestions = []
        page_width, page_height = page_size
        
        # A4 기준 템플릿 (210x297mm)
        if 200 <= page_width <= 220 and 285 <= page_height <= 305:
            # 2-up on A3
            suggestions.append(ImpositionTemplate(
                name="2-up A4 on A3",
                sheet_width=420,
                sheet_height=297,
                rows=1,
                columns=2,
                gutter=5
            ))
            
            # 4-up on A2
            if page_count >= 8:
                suggestions.append(ImpositionTemplate(
                    name="4-up A4 on A2",
                    sheet_width=594,
                    sheet_height=420,
                    rows=2,
                    columns=2,
                    gutter=5
                ))
            
            # 중철 제본용
            if page_count >= 8 and page_count % 4 == 0:
                suggestions.append(ImpositionTemplate(
                    name="Saddle-stitch A5 booklet",
                    sheet_width=420,
                    sheet_height=297,
                    rows=1,
                    columns=2,
                    binding_method=BindingMethod.SADDLE_STITCH
                ))
        
        return suggestions
    
    def _load_standard_templates(self) -> List[ImpositionTemplate]:
        """표준 템플릿 로드"""
        # 실제로는 JSON 파일에서 로드
        return []
    
    def can_analyze(self, document: PDFDocument) -> bool:
        """항상 분석 가능"""
        return True
```

```python
from ..core.models import PDFDocument, AnalysisResult, QualityIssue
from ..core.analyzers import *
from ..core.checkers import *

@dataclass
class PipelineOptions:
    """파이프라인 처리 옵션"""
    profile_name: str = "default"
    auto_fix: bool = False
    fix_options: Dict[str, bool] = None
    generate_report: bool = True
    report_formats: List[str] = None

    check_imposition_readiness: bool = False
    suggest_imposition_templates: bool = False

        if self.fix_options is None:
            self.fix_options = {}
        if self.report_formats is None:
            self.report_formats = ['html']

class PDFProcessingPipeline:
    """PDF 처리 파이프라인 - 모든 처리의 중심"""
    
    def __init__(self, tool_manager, profile_manager):
        self.tool_manager = tool_manager
        self.profile_manager = profile_manager
        
        # 분석기 초기화
        self.analyzers = [
            MetadataAnalyzer(),
            PageAnalyzer(),
            FontAnalyzer(tool_manager),
            ColorAnalyzer(),
            ImageAnalyzer()
        ]
        
        # 판짜기 준비 분석기 (선택적)
        self.imposition_analyzer = ImpositionReadinessAnalyzer()
        
        # 검사기 초기화
        self.checkers = [
            TransparencyChecker(),
            OverprintChecker(tool_manager),
            BleedChecker(),
            SpotColorChecker(),
            InkCoverageChecker()
        ]
        
        # 수정기 초기화
        self.fixers = {
            'color': ColorFixer(tool_manager),
            'font': FontFixer(tool_manager)
        }
        
        # 판짜기 엔진 (향후 구현)
        self.imposition_engine = None  # 나중에 ImpositionEngine()
    
    def process(self, pdf_path: Path, options: PipelineOptions) -> Dict[str, Any]:
        """PDF 파일 처리 - 전체 워크플로우"""
        
        # 1. 프로파일 로드
        profile = self.profile_manager.load_profile(options.profile_name)
        
        # 2. 문서 객체 생성
        document = PDFDocument.from_path(pdf_path)
        
        # 3. 분석 수행
        analysis_result = self._analyze(document, profile)
        
        # 4. 품질 검사
        quality_issues = self._check_quality(analysis_result, profile)
        analysis_result.issues.extend(quality_issues)
        
        # 5. 판짜기 준비 상태 확인 (옵션)
        if options.check_imposition_readiness:
            imposition_info = self._check_imposition_readiness(document, analysis_result)
            analysis_result.imposition_readiness = imposition_info
        
        # 6. 자동 수정 (옵션)
        fix_result = None
        if options.auto_fix and analysis_result.has_errors:
            fix_result = self._auto_fix(document, analysis_result, options.fix_options)
        
        # 7. 보고서 생성 (옵션)
        reports = {}
        if options.generate_report:
            reports = self._generate_reports(analysis_result, fix_result, options.report_formats)
        
        return {
            'document': document,
            'analysis': analysis_result,
            'fix_result': fix_result,
            'reports': reports,
            'imposition_ready': analysis_result.is_imposition_ready if options.check_imposition_readiness else None
        }
    
    def _check_imposition_readiness(self, document: PDFDocument, analysis: AnalysisResult) -> 'ImpositionReadiness':
        """판짜기 준비 상태 확인"""
        context = {
            'pages': [page.__dict__ for page in analysis.pages],
            'profile': {}
        }
        
        result = self.imposition_analyzer.analyze(document, context)
        return result.get('imposition_readiness')
```
